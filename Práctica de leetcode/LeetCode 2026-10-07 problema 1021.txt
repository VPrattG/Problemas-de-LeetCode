public class Solution {
    public string RemoveOuterParentheses(string s) {
        int depth = 0;
        StringBuilder sb = new StringBuilder();

        foreach (char c in s) {
            if (c == ')') depth--;
            if (depth > 0) sb.Append(c);
            if (c == '(') depth++;
        }

        return sb.ToString();
    }
}
