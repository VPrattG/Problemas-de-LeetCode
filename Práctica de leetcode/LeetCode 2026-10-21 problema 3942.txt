public class Solution {
    public int MinOperations(int[] nums) {
        int n = nums.Length;
        if (n == 1) return 0;

        int start = Array.IndexOf(nums, 0);
        bool asc = true;
        bool des = true;

        for (int i = 1; i < n; i++) {
            int value = nums[(start + i) % n];

            // Solo puede funcionar si está en orden ascendente o descendente
            if (value != i) asc = false;
            if (value != n - i) des = false;

            if (!asc && !des) return -1;
        }
        
        // Se revisa cómo es más rápido llevar el 0
        // Girando a la izquierda, o inviritiendo, girando, e invertir de nuevo
        if (asc) return Math.Min(start, n - start + 2);

        return Math.Min(n - start, start + 2);
    }
}
