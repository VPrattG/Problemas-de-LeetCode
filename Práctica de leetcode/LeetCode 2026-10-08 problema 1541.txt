public class Solution {
    public int MinInsertions(string s) {
        int openBrackets = 0, minAdds = 0;
        int length = s.Length, index = 0;

        while (index < length) {
            char c = s[index];
            if (c == '(') {
                openBrackets++; index++;
            } else {
                if (openBrackets > 0) openBrackets--;
                else minAdds++;

                if (index < length - 1 && s[index + 1] == ')')
                    index += 2;
                else {
                    minAdds++;
                    index++;
                }
            }
        }

        minAdds += openBrackets * 2;
        return minAdds;
    }
}
