public class Solution {
    public int MinMoves(int[] nums) {
        int max = nums[0];
        for (int i = 1; i < nums.Length; i++) {
            max = Math.Max(max, nums[i]);
        }

        int result = 0;
        foreach (int num in nums) {
            result += max - num;
        }

        return result;
    }
}
