public class Solution {
    public string SmallestPalindrome(string s) {
        int n = s.Length;
        int[] letters = new int[26];

        for(int i = 0; i < n / 2; i++){
            letters[s[i] - 'a']++;
        }

        char[] result = new char[n];
        int start = 0, end = n - 1;

        for(int i = 0; i < 26; i++){
            while(letters[i] > 0){
                char c = (char)('a' + i);
                result[start++] = c;
                result[end--] = c;
                letters[i]--;
            }
        }

        if(n % 2 != 0){
            result[start] = s[n / 2];
        }

        return new string(result);
    }
}
