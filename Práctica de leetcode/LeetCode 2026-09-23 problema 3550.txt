public class Solution {
    public int SmallestIndex(int[] nums) {
        for(int i = 0; i < nums.Length; i++){
            int num = nums[i];
            int digitSum = 0;

            while(num > 0){
                digitSum += num % 10;
                num /= 10;
            }

            if(i == digitSum) return i;
        }

        return -1;
    }
}
