public class Solution {
    public int AuxGCD(int a, int b){
        while(b != 0){
            int rem = a % b;
            a = b;
            b = rem;
        }

        return a;
    }

    public long GcdSum(int[] nums) {
        int n = nums.Length;
        int preMax = int.MinValue;
        
        int[] max = new int[n];
        int[] prefixGcd = new int[n];

        for(int i = 0; i < n; i++){
            preMax = Math.Max(preMax, nums[i]);
            max[i] = preMax;
        }
        
        for(int i = 0; i < n; i++){
            prefixGcd[i] = AuxGCD(nums[i], max[i]);
        }

        Array.Sort(prefixGcd);

        long result = 0;
        int left = 0, right = n - 1;

        while(left < right){
            result += AuxGCD(prefixGcd[left], prefixGcd[right]);
            left++; right--;
        }

        return result;
    }
}
