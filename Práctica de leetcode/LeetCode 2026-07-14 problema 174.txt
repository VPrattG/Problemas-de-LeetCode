public class Solution {
    public int CalculateMinimumHP(int[][] dungeon) {
        int n = dungeon.Length;
        int m = dungeon[0].Length;

        int[][] dp = new int[n + 1][];

        for(int i = 0; i <= n; i++){
            dp[i] = new int[m + 1];
            Array.Fill(dp[i], -1);
        }

        return DungeonSolver(0, 0, n, m, dungeon, dp);
    }

    public int DungeonSolver(int i, int j, int n, int m, int[][] dungeon, int[][] dp){
        if(i >= n || j >= m) return int.MaxValue;

        if(i == n - 1 && j == m - 1)
            return dungeon[i][j] < 0 ? Math.Abs(dungeon[i][j]) + 1 : 1;

        if(dp[i][j] != -1) return dp[i][j];

        int down = DungeonSolver(i + 1, j, n, m, dungeon, dp);
        int right = DungeonSolver(i, j + 1, n, m, dungeon, dp);

        int needed = Math.Min(down, right) - dungeon[i][j];
        dp[i][j] = (needed <= 0) ? 1 : needed;
        return dp[i][j];
    }
}
