public class Solution {
    public string FractionToDecimal(int numerator, int denominator) {
        if(numerator == 0) return "0";

        StringBuilder fraction = new StringBuilder();
        // if puede usar XOR
        if(numerator < 0 ^ denominator < 0)
            fraction.Append("-");

        // Evita el desbordamiento de datos con int
        long dividend = Math.Abs((long)numerator);
        long divisor = Math.Abs((long)denominator);

        fraction.Append(dividend / divisor);

        long remainder = dividend % divisor;

        if(remainder == 0) return fraction.ToString();

        fraction.Append(".");

        Dictionary<long, int> dic = new Dictionary<long, int>();

        while(remainder != 0){
            if(dic.ContainsKey(remainder)){
                fraction.Insert(dic[remainder], "(");
                fraction.Append(")");
                break;
            }

            // Se almacena el valor donde se debe insertar el paréntesis
            dic[remainder] = fraction.Length;
            remainder *= 10;
            fraction.Append(remainder / divisor);
            remainder %= divisor;
        }

        return fraction.ToString();
    }
}
