public class Solution {
    public int MinSumOfLengths(int[] arr, int target) {
        // Permite búsquedas en O(n)
        Dictionary<int, int> position = new Dictionary<int, int>();
        position[0] = -1;
        int n = arr.Length, sum = 0, result = n + 1, minLength = n;

        for(int i = 0; i < n; i++){
            sum += arr[i];
            // Revisa si el subarreglo actual equivale a target
            // Si target = 7, arr[i] es 21, y existe una llave en position con un valor de 14
            // significa que arr[j + 1 ... i] tiene un valor de 7
            if(position.TryGetValue(sum - target, out int j)){
                // Determina si la longitud del subarreglo actual es la nueva longitud mínima
                int length = i - j;
                // Obtiene la longitud mínima en la posición j
                result = Math.Min(result, length + (j == -1 ? n : arr[j]));
                minLength = Math.Min(minLength, length);
            }
            // Almacena la longitud mínima hasta ese punto
            arr[i] = minLength;
            // Guarda la suma actual y su posición en el arreglo
            position[sum] = i;
        }

        return result == n + 1 ? -1 : result;
    }
}
