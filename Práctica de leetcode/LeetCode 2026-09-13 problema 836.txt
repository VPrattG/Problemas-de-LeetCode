public class Solution {
    public bool IsRectangleOverlap(int[] rec1, int[] rec2) {
        // Revisa si uno de los rectángulos es una línea
        if(rec1[0] == rec1[2] || rec1[1] == rec1[3] ||
            rec2[0] == rec2[2] || rec2[1] == rec2[3])
                return false;

        // Revisa si rec1 está a la izquierda, abajo, 
        // derecha o arriba de rec2
        return !(rec1[2] <= rec2[0] ||
                 rec1[3] <= rec2[1] ||
                 rec1[0] >= rec2[2] ||
                 rec1[1] >= rec2[3]);
    }
}
