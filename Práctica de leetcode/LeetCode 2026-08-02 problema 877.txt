public class Solution {
    public bool StoneGame(int[] piles) {
        int n = piles.Length;
        int[] dp = new int[n];

        for(int i = n - 1; i >= 0; i--){
            dp[i] = piles[i];
            for(int j = i + 1; j < n; j++){
                dp[j] = Math.Max(piles[i] - dp[j], piles[j] - dp[j - 1]);
            }
        }

        return dp[n - 1] > 0;
    }
}
