
public class Solution {
    public int MaxActiveSectionsAfterTrade(string s) {
        int n = s.Length;
        int count1 = 0;

        foreach(char c in s){
            if(c == '1') count1++;
        }

        int i = 0;
        int highest = 0;
        int prev = int.MinValue;
        int cur = 0;
        
        while(i < n){
            int start = i;
            
            while(i < n && s[i] == s[start]) i++;

            if(s[start] == '0'){
                cur = i - start;

                if(prev != int.MinValue){
                    highest = Math.Max(highest, prev + cur);
                }

                prev = cur;
            }
        }

        return count1 + highest;
    }
}
