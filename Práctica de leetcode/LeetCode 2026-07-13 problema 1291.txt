public class Solution {
    public static int[] validNums = new int[45];

    static Solution(){
        int n = 0;
        for(int i = 1; i < 10; i++){
            validNums[n++] = i;
        }

        for(int i = 0; i < n; i++){
            int digit = validNums[i] % 10;
            if(digit < 9) validNums[n++] = validNums[i] * 10 + digit + 1;
        }
    }

    public IList<int> SequentialDigits(int low, int high) {
        IList<int> result = new List<int>();

        foreach(int num in validNums)
            if(num >= low && num <= high)
                result.Add(num);

        return result;
    }
}
