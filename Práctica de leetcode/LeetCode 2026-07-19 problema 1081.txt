public class Solution {
    public string SmallestSubsequence(string s) {
        int[] freq = new int[27];
        bool[] seen = new bool[27];
        Stack<char> stack = new Stack<char>();

        for(int i = 0; i < s.Length; i++){
            freq[s[i] - 'a']++;
            Console.WriteLine(s[i]);
        }

        for(int i = 0; i < s.Length; i++){
            char c = s[i];
            int position = c - 'a';
            freq[position]--;

            if(seen[position]) continue;

            while(stack.Count > 0){
                if(stack.Peek() <= c) break;

                if(freq[stack.Peek() - 'a'] == 0) break;

                seen[stack.Peek() - 'a'] = false;

                stack.Pop();
            }

            stack.Push(c);
            seen[position] = true;
        }

        StringBuilder sb = new StringBuilder();

        foreach(char c in stack) sb.Insert(0, c);

        return sb.ToString();
    }
}
