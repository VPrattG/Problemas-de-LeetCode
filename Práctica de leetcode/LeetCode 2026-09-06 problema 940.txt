public class Solution {
    public int DistinctSubseqII(string s) {
        int modulo = 1000000007;
        int n = s.Length;
        int[] dp = new int[n + 1];
        dp[0] = 1;

        int[] last = new int[26];
        for(int i = 0; i < last.Length; i++) last[i] = -1;

        for(int i = 0; i < n; i++){
            int x = s[i] - 'a';
            // Agrega la letra actual a la lista de combinaciones anteriores
            dp[i + 1] = dp[i] * 2 % modulo;
            // Por si se encontró la letra anteriormente
            if(last[x] >= 0) dp[i + 1] -= dp[last[x]];
            dp[i + 1] %= modulo;
            last[x] = i;
        }

        // Remueve cadena vacía
        dp[n]--;
        if(dp[n] < 0) dp[n] += modulo;
        return dp[n];
    }
}
