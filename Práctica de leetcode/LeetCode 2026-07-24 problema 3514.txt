public class Solution {
    public int UniqueXorTriplets(int[] nums) {
        int max = 0;
        foreach(int num in nums){
            max = Math.Max(max, num);
        }

        // Busca la potencia de 2 más pequeña, mayor a max
        int u = 1;
        while(u <= max){
            u <<= 1;
        }

        bool[] one = new bool[u];
        bool[] two = new bool[u];
        bool[] three = new bool[u];
        foreach(int v in nums){
            one[v] = true;
            for(int x = 0; x < u; x++){
                if(one[x]){
                    two[x^v] = true;
                }
            }
        }

        foreach(int v in nums){
            for(int x = 0; x < u; x++){
                if(two[x]){
                    three[x^v] = true;
                }
            }
        }

        int result = 0;
        foreach(bool b in three){
            if(b) result++;
        }

        return result;
    }
}
