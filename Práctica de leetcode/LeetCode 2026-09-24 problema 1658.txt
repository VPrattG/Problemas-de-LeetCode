public class Solution {
    public int MinOperations(int[] nums, int x) {
        int k = -x, n = nums.Length;

        foreach(int num in nums){
            k += num;
        }
        
        if(k < 0) return -1;
        if(k == 0) return n;

        int best = -1, i = 0, sum = 0;

        // i  es el inicio de la ventana, y j el final
        for(int j = 0; j < n; j++){
            sum += nums[j];
            while(sum > k){
                sum -= nums[i++];
            }

            if(sum == k){
                // Al encontrar el subarreglo más grande que equivale a 0
                // encontramos la menor cantidad de pasos para reducir el arreglo
                best = Math.Max(best, j - i + 1);
            }
        }

        return best < 0 ? -1 : n - best;
    }
}
