public class Solution {
    public int LongestMonotonicSubarray(int[] nums) {
        int maxL = 1, incL = 1, decL = 1;

        for (int i = 0; i < nums.Length - 1; i++) {
            if (nums[i + 1] > nums[i]) {
                incL++; decL = 1;
            } else if (nums[i + 1] < nums[i]) {
                incL = 1; decL++;
            } else {
                incL = 1; decL = 1;
            }

            maxL = Math.Max(maxL, Math.Max(incL, decL));
        }

        return maxL;
    }
}
