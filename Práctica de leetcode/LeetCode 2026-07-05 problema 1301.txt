public class Solution {
    private const int MODULO = 1000000007;
    
    public int[] PathsWithMaxScore(IList<string> board) {
        int n = board.Count;

        // Valores de la fila inferior
        int[] nextScore = new int[n + 1];
        int[] nextWays = new int[n + 1];

        for(int i = 0; i < nextScore.Length; i++){
            nextScore[i] = -1;
        }

        for(int i = n - 1; i >= 0; i--){
            int[] currScore = new int[n + 1];
            int[] currWays = new int[n + 1];

            for(int j = 0; j < currScore.Length; j++){
                currScore[j] = -1;
            }

            for(int j = n - 1; j >= 0; j--){
                char cell = board[i][j];

                if(cell == 'X') continue;

                // Solo para la casilla inicial
                if(cell == 'S'){
                    currScore[j] = 0;
                    currWays[j] = 1;
                    continue;
                }

                int best = Math.Max(
                    nextScore[j], 
                    Math.Max(currScore[j + 1], nextScore[j + 1])
                );

                if(best == -1) continue;

                long ways = 0;

                if(nextScore[j] == best) ways += nextWays[j];
                if(currScore[j + 1] == best) ways += currWays[j + 1];
                if(nextScore[j + 1] == best) ways += nextWays[j + 1];

                int value = cell == 'E' ? 0 : cell - '0';

                currScore[j] = best + value;
                currWays[j] = (int) (ways % MODULO);
            }

            nextScore = currScore;
            nextWays = currWays;
        }

        if(nextScore[0] == -1) return new int[] {0,0};

        return new int[]{nextScore[0], nextWays[0]};
    }
}
