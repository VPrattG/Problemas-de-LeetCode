public class Solution {
    public IList<int> FindMissingElements(int[] nums) {
        Array.Sort(nums);
        IList<int> result = new List<int>();
        
        for(int i = 0; i < nums.Length - 1; i++){
            for(int j = nums[i] + 1; j < nums[i + 1]; j++){
                result.Add(j);
            }
        }

        return result;
    }
}
