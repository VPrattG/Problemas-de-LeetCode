public class Solution {
    public int[] ResultArray(int[] nums) {
        int n = nums.Length;
        int[] result = new int[n];
        result[0] = nums[0];
        result[n - 1] = nums[1];
        int index = 0, revIndex = n - 1;

        for(int i = 2; i < n; i++){
            if(result[index] > result[revIndex])
                result[++index] = nums[i];
            else
                result[--revIndex] = nums[i];
        }

        Array.Reverse(result, revIndex, n - revIndex);
        return result;
    }
}
