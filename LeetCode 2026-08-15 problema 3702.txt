public class Solution {
    public int LongestSubsequence(int[] nums) {
        int n = nums.Length;
        int totalXor = 0;
        bool allZero = true;

        foreach(int num in nums){
            totalXor ^= num;
            if(num > 0){
                allZero = false;
            }
        }

        if(totalXor > 0) return n;

        return allZero ? 0 : n - 1;
    }
}
